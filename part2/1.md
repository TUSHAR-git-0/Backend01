<!-- we can have two type of middlewares  -->

<!-- global   :   which runs every times   -> no matters on which route we are  going -->
<!-- route specific  :   which runs only on specific route  -->
<!-- router level middlewares -->


<!-- 1. global middlewares  -->    app.use(express.json()); 



<!-- 2. route specific middlewares  -->    jab hum kisi specific route pe middleware use karna chahte hai to hum us route ke sath us middleware ko specify kar sakte hai  -->


like : 


function middleware1 ( req , res , next ) {
    // code for middleware1
    next();  // call the next middleware or route handler
}

function middleware2 ( req , res , next ) {
    // code for middleware2
    next();  // call the next middleware or route handler
}

app.get('books', middleware1, middleware2, (req, res) => {
    // code to handle the request
});

<!-- ab jab bhi humme route p  get  milta hai to   , middleware1   then   middleware2  run karega   -->

**Har route p  jo hum ye function likhte hai , ye bhi humara middleware hi hai !**


**Actually ( req, res )={  }    on every route --->    ye bhi ek middleware hai  , jisme humne next() nahi diya hai  , isliye ye last middleware hai  , aur ye response bhejta hai client ko**


issme hum        next()        p nhi jaa rhe because  : ye humara last middleware ki trh treate ho rha hai , returning the response to client
